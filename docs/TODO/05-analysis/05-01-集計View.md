## タスク詳細

### 1. 診断結果サマリView

**ファイル**: `supabase/migrations/YYYYMMDD_analysis_views.sql`

**View名**: `diagnosis_summary_view`

```sql
CREATE OR REPLACE VIEW diagnosis_summary_view AS
SELECT 
  e.id as event_id,
  e.name as event_name,
  e.event_date,
  dc.name as category_name,
  di.question,
  di.answer_type,
  dr.value,
  COUNT(*) as count,
  ROUND(
    COUNT(*) * 100.0 / 
    SUM(COUNT(*)) OVER (PARTITION BY e.id, di.id), 
    1
  ) as percentage
FROM diagnosis_responses dr
JOIN diagnosis_items di ON dr.item_id = di.id
JOIN diagnosis_categories dc ON di.category_id = dc.id
JOIN sessions s ON dr.session_id = s.session_id
LEFT JOIN visits v ON v.session_id = s.session_id
LEFT JOIN events e ON v.event_id = e.id
WHERE s.status = 'completed'
GROUP BY e.id, e.name, e.event_date, dc.name, di.question, di.answer_type, dr.value
ORDER BY e.event_date DESC, dc.display_order, di.display_order, count DESC;
```

### 2. 問診結果サマリView

**View名**: `questionnaire_summary_view`

```sql
CREATE OR REPLACE VIEW questionnaire_summary_view AS
SELECT 
  e.id as event_id,
  e.name as event_name,
  qc.name as category_name,
  qc.target_age,
  qi.question,
  qi.answer_type,
  qr.value,
  COUNT(*) as count,
  ROUND(
    COUNT(*) * 100.0 / 
    SUM(COUNT(*)) OVER (PARTITION BY e.id, qi.id), 
    1
  ) as percentage
FROM questionnaire_responses qr
JOIN questionnaire_items qi ON qr.item_id = qi.id
JOIN questionnaire_categories qc ON qi.category_id = qc.id
JOIN sessions s ON qr.session_id = s.session_id
LEFT JOIN visits v ON v.session_id = s.session_id
LEFT JOIN events e ON v.event_id = e.id
GROUP BY e.id, e.name, qc.name, qc.target_age, qi.question, qi.answer_type, qr.value
ORDER BY qc.display_order, qi.display_order, count DESC;
```

### 3. イベント別統計View

**View名**: `event_statistics_view`

```sql
CREATE OR REPLACE VIEW event_statistics_view AS
SELECT 
  e.id as event_id,
  e.name as event_name,
  e.event_date,
  e.venue,
  COUNT(DISTINCT s.session_id) as total_sessions,
  COUNT(DISTINCT CASE WHEN s.status = 'completed' THEN s.session_id END) as completed_sessions,
  COUNT(DISTINCT dr.session_id) as diagnosed_sessions,
  COUNT(DISTINCT qr.session_id) as questionnaire_sessions,
  ROUND(
    COUNT(DISTINCT CASE WHEN s.status = 'completed' THEN s.session_id END) * 100.0 / 
    NULLIF(COUNT(DISTINCT s.session_id), 0),
    1
  ) as completion_rate
FROM events e
LEFT JOIN visits v ON v.event_id = e.id
LEFT JOIN sessions s ON v.session_id = s.session_id
LEFT JOIN diagnosis_responses dr ON dr.session_id = s.session_id
LEFT JOIN questionnaire_responses qr ON qr.session_id = s.session_id
GROUP BY e.id, e.name, e.event_date, e.venue
ORDER BY e.event_date DESC;
```

### 4. 日別診断数View

**View名**: `daily_diagnosis_count_view`

```sql
CREATE OR REPLACE VIEW daily_diagnosis_count_view AS
SELECT 
  DATE(s.created_at) as date,
  e.name as event_name,
  COUNT(DISTINCT s.session_id) as session_count,
  COUNT(DISTINCT CASE WHEN s.status = 'completed' THEN s.session_id END) as completed_count
FROM sessions s
LEFT JOIN visits v ON v.session_id = s.session_id
LEFT JOIN events e ON v.event_id = e.id
GROUP BY DATE(s.created_at), e.name
ORDER BY date DESC;
```

### 5. スタッフ別診断数View

**View名**: `staff_diagnosis_count_view`

```sql
CREATE OR REPLACE VIEW staff_diagnosis_count_view AS
SELECT 
  p.id as staff_id,
  p.display_name as staff_name,
  e.name as event_name,
  COUNT(DISTINCT v.session_id) as diagnosis_count,
  AVG(
    EXTRACT(EPOCH FROM (
      (SELECT MAX(answered_at) FROM diagnosis_responses WHERE session_id = v.session_id) -
      v.visit_date
    )) / 60
  ) as avg_diagnosis_minutes
FROM visits v
JOIN profiles p ON v.staff_profile_id = p.id
LEFT JOIN events e ON v.event_id = e.id
GROUP BY p.id, p.display_name, e.name
ORDER BY diagnosis_count DESC;
```

### 6. 項目別異常検出View

**View名**: `abnormal_findings_view`

```sql
CREATE OR REPLACE VIEW abnormal_findings_view AS
SELECT 
  s.session_id,
  s.parent_name,
  dc.name as category_name,
  di.question,
  dr.value,
  dr.answered_at
FROM diagnosis_responses dr
JOIN diagnosis_items di ON dr.item_id = di.id
JOIN diagnosis_categories dc ON di.category_id = dc.id
JOIN sessions s ON dr.session_id = s.session_id
WHERE 
  -- 異常値のパターン（例）
  (di.question = '舌小帯短縮症' AND dr.value = 'yes')
  OR (di.question = '低位舌' AND dr.value = 'yes')
  OR (di.question = '口呼吸・鼻呼吸' AND dr.value = 'mouth')
  OR (di.question = '反対咬合' AND dr.value = 'yes')
ORDER BY dr.answered_at DESC;
```
