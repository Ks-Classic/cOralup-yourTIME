## 概要

親御さんの問診画面をLIFF（LINE Front-end Framework）で実装。
ブラウザ閉じても復元可能、LINE User ID確実取得、機種依存軽減。

### なぜLIFFか

| 観点 | 通常Web | LIFF |
|------|---------|------|
| LINE User ID取得 | URLパラメータ（改ざんリスク） | `liff.getProfile()`（確実） |
| 途中離脱からの復元 | Cookie/LocalStorage依存 | LINE User IDでDB検索→復元 |
| 機種依存 | ブラウザ差異あり | LINEのWebViewで統一 |
| ユーザー離脱 | ブラウザ閉じると迷子 | LINEアプリ内で完結 |

## 実装詳細

### 2.2 LIFF問診ページ

```typescript
// src/app/(exhibition)/parent/questionnaire/liff/page.tsx
'use client'

import { useEffect, useState } from 'react'
import liff from '@line/liff'

export default function LiffQuestionnairePage() {
  const [isInitialized, setIsInitialized] = useState(false)
  const [lineUserId, setLineUserId] = useState<string | null>(null)
  const [existingVisit, setExistingVisit] = useState<Visit | null>(null)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    const initLiff = async () => {
      try {
        await liff.init({ liffId: process.env.NEXT_PUBLIC_PARENT_LIFF_ID! })

        // 外部ブラウザで開かれた場合
        if (!liff.isInClient()) {
          // LINEアプリへ誘導
          setError('LINEアプリ内で開いてください')
          return
        }

        // ログインしていない場合
        if (!liff.isLoggedIn()) {
          liff.login({ redirectUri: window.location.href })
          return
        }

        // プロフィール取得
        const profile = await liff.getProfile()
        setLineUserId(profile.userId)

        // 既存visitを検索
        const res = await fetch(`/api/parent/visit?line_user_id=${profile.userId}`)
        const data = await res.json()
        
        if (data.visit) {
          setExistingVisit(data.visit)
        }

        setIsInitialized(true)
      } catch (err) {
        console.error('LIFF init error:', err)
        setError('初期化に失敗しました')
      }
    }

    initLiff()
  }, [])

  // ... 問診フォーム表示
}
```

### 2.5 既存visit復元API

```typescript
// src/app/api/parent/visit/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { createServerSupabaseClient } from '@/lib/supabase'

export async function GET(request: NextRequest) {
  const { searchParams } = new URL(request.url)
  const lineUserId = searchParams.get('line_user_id')

  if (!lineUserId) {
    return NextResponse.json({ error: 'line_user_id is required' }, { status: 400 })
  }

  const supabase = createServerSupabaseClient()

  // 1. profilesからparent_profile_id取得
  const { data: profile } = await supabase
    .from('profiles')
    .select('id')
    .eq('line_user_id', lineUserId)
    .eq('role', 'parent')
    .single()

  if (!profile) {
    return NextResponse.json({ visit: null })
  }

  // 2. 未完了のvisitを検索
  const { data: visit } = await supabase
    .from('visits')
    .select(`
      id,
      status,
      session_id,
      children (
        id,
        first_name,
        last_name,
        birthday,
        gender
      )
    `)
    .eq('children.parent_profile_id', profile.id)
    .in('status', ['waiting', 'questionnaire_in_progress'])
    .order('created_at', { ascending: false })
    .limit(1)
    .single()

  // 3. 問診回答も取得
  let questionnaireResponses = []
  if (visit?.session_id) {
    const { data: responses } = await supabase
      .from('questionnaire_responses')
      .select('item_id, value')
      .eq('session_id', visit.session_id)

    questionnaireResponses = responses || []
  }

  return NextResponse.json({
    visit,
    questionnaireResponses,
  })
}
```
