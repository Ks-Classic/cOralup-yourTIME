# 画像アップロード仕様書 (cOralup Platform)

## 1. 概要

診断時に撮影する写真（姿勢・口腔内）をSupabase Storageにアップロードし、管理する。

---

## 2. Supabase Storage バケット設計

### 2.1 バケット構成

| バケット名 | 用途 | 公開設定 | RLS |
|-----------|------|---------|-----|
| `diagnosis-photos` | 診断写真（姿勢・口腔内） | Private | あり |
| `avatars` | スタッフ・ユーザーアイコン | Public | あり |

### 2.2 ディレクトリ構造

```
diagnosis-photos/
├── {event_id}/
│   ├── {visit_id}/
│   │   ├── posture_front.jpg    # 正面姿勢
│   │   ├── posture_side.jpg     # 横向き姿勢
│   │   ├── oral_front.jpg       # 口腔内（正面）
│   │   ├── oral_upper.jpg       # 口腔内（上顎）
│   │   └── oral_lower.jpg       # 口腔内（下顎）
│   └── ...
└── ...

avatars/
├── {profile_id}.jpg
└── ...
```

### 2.3 SQL設定

```sql
-- バケット作成
INSERT INTO storage.buckets (id, name, public)
VALUES 
  ('diagnosis-photos', 'diagnosis-photos', false),
  ('avatars', 'avatars', true);

-- RLSポリシー: diagnosis-photos
-- スタッフのみアップロード可能
CREATE POLICY "Staff can upload diagnosis photos"
ON storage.objects FOR INSERT
TO authenticated
WITH CHECK (
  bucket_id = 'diagnosis-photos'
  AND EXISTS (
    SELECT 1 FROM profiles
    WHERE id = auth.uid()
    AND role IN ('staff', 'admin')
  )
);

-- 認証済みユーザーは閲覧可能
CREATE POLICY "Authenticated users can view diagnosis photos"
ON storage.objects FOR SELECT
TO authenticated
USING (bucket_id = 'diagnosis-photos');

-- RLSポリシー: avatars
-- 誰でも閲覧可能
CREATE POLICY "Anyone can view avatars"
ON storage.objects FOR SELECT
TO public
USING (bucket_id = 'avatars');

-- 本人のみアップロード可能
CREATE POLICY "Users can upload own avatar"
ON storage.objects FOR INSERT
TO authenticated
WITH CHECK (
  bucket_id = 'avatars'
  AND (storage.foldername(name))[1] = auth.uid()::text
);
```

## 4. アップロード仕様

### 4.1 制限

| 項目 | 制限値 |
|------|--------|
| 最大ファイルサイズ | **10MB** |
| 許可形式 | `image/jpeg`, `image/png`, `image/webp` |
| 最大解像度 | 4096 x 4096 px |
| 1診断あたり最大枚数 | **10枚** |

### 4.2 画像処理

```typescript
// アップロード前の処理
const processImage = async (file: File): Promise<Blob> => {
  // 1. EXIF回転の正規化
  const oriented = await fixOrientation(file)
  
  // 2. リサイズ（長辺1920px以下）
  const resized = await resizeImage(oriented, { maxWidth: 1920, maxHeight: 1920 })
  
  // 3. JPEG圧縮（品質85%）
  const compressed = await compressImage(resized, { quality: 0.85 })
  
  return compressed
}
```

## 5. 実装コード

### 5.1 アップロードフック

```typescript
// src/hooks/usePhotoUpload.ts
import { createClient } from '@supabase/supabase-js'

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
)

interface UploadResult {
  success: boolean
  url?: string
  error?: string
}

export function usePhotoUpload() {
  const uploadPhoto = async (
    file: File,
    eventId: string,
    visitId: string,
    photoType: string
  ): Promise<UploadResult> => {
    // バリデーション
    if (file.size > 10 * 1024 * 1024) {
      return { success: false, error: 'ファイルサイズが10MBを超えています' }
    }
    
    if (!['image/jpeg', 'image/png', 'image/webp'].includes(file.type)) {
      return { success: false, error: '対応していない画像形式です' }
    }
    
    try {
      // 画像処理
      const processed = await processImage(file)
      
      // ファイル名生成
      const timestamp = Date.now()
      const extension = file.type === 'image/png' ? 'png' : 'jpg'
      const fileName = `${eventId}/${visitId}/${photoType}_${timestamp}.${extension}`
      
      // アップロード
      const { data, error } = await supabase.storage
        .from('diagnosis-photos')
        .upload(fileName, processed, {
          contentType: file.type,
          upsert: false
        })
      
      if (error) throw error
      
      // 署名付きURL取得（1時間有効）
      const { data: urlData } = await supabase.storage
        .from('diagnosis-photos')
        .createSignedUrl(data.path, 3600)
      
      return { success: true, url: urlData?.signedUrl }
      
    } catch (error) {
      console.error('Upload error:', error)
      return { success: false, error: 'アップロードに失敗しました' }
    }
  }
  
  return { uploadPhoto }
}
```

### 5.2 カメラ撮影コンポーネント

```tsx
// src/components/staff/PhotoCapture.tsx
'use client'

import { useRef, useState } from 'react'
import { usePhotoUpload } from '@/hooks/usePhotoUpload'

interface PhotoCaptureProps {
  eventId: string
  visitId: string
  photoType: string
  label: string
  onUpload: (url: string) => void
}

export function PhotoCapture({ 
  eventId, 
  visitId, 
  photoType, 
  label,
  onUpload 
}: PhotoCaptureProps) {
  const inputRef = useRef<HTMLInputElement>(null)
  const [preview, setPreview] = useState<string | null>(null)
  const [uploading, setUploading] = useState(false)
  const { uploadPhoto } = usePhotoUpload()

  const handleCapture = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return
    
    // プレビュー表示
    const reader = new FileReader()
    reader.onload = (e) => setPreview(e.target?.result as string)
    reader.readAsDataURL(file)
    
    // アップロード
    setUploading(true)
    const result = await uploadPhoto(file, eventId, visitId, photoType)
    setUploading(false)
    
    if (result.success && result.url) {
      onUpload(result.url)
    } else {
      alert(result.error || 'アップロードに失敗しました')
    }
  }

  return (
    <div className="flex flex-col items-center">
      <input
        ref={inputRef}
        type="file"
        accept="image/*"
        capture="environment"
        onChange={handleCapture}
        className="hidden"
      />
      
      {preview ? (
        <div className="relative">
          <img 
            src={preview} 
            alt={label}
            className="w-48 h-48 object-cover rounded-lg"
          />
          {uploading && (
            <div className="absolute inset-0 bg-black/50 flex items-center justify-center rounded-lg">
              <span className="text-white">アップロード中...</span>
            </div>
          )}
        </div>
      ) : (
        <button
          onClick={() => inputRef.current?.click()}
          className="w-48 h-48 border-2 border-dashed border-gray-300 rounded-lg flex flex-col items-center justify-center text-gray-500 hover:border-blue-500 hover:text-blue-500"
        >
          <span className="text-4xl mb-2">📷</span>
          <span>{label}</span>
        </button>
      )}
    </div>
  )
}
```

// 一時的なアクセスURL（有効期限付き）
const getSignedUrl = async (path: string, expiresIn: number = 3600) => {
  const { data } = await supabase.storage
    .from('diagnosis-photos')
    .createSignedUrl(path, expiresIn)
  return data?.signedUrl
}

## 7. レポート表示時の画像取得

```typescript
// src/app/report/[id]/page.tsx
const getPhotoUrls = async (visitId: string) => {
  const { data: diagnosis } = await supabase
    .from('oral_diagnoses')
    .select('photo_urls')
    .eq('visit_id', visitId)
    .single()
  
  if (!diagnosis?.photo_urls) return []
  
  // 署名付きURLを生成（レポート閲覧用に長めの有効期限）
  const signedUrls = await Promise.all(
    diagnosis.photo_urls.map(async (path: string) => {
      const { data } = await supabase.storage
        .from('diagnosis-photos')
        .createSignedUrl(path, 86400) // 24時間
      return data?.signedUrl
    })
  )
  
  return signedUrls.filter(Boolean)
}
```
